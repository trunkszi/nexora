cmake_minimum_required(VERSION 4.0)
include(cmake/NexoraToolchain.cmake)
nexora_ensure_toolchain()
project(nexora VERSION 0.1.0 DESCRIPTION "C++26 Sans-IO protocols and Proactor transport" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(NEXORA_BUILD_TESTS "Build deterministic tests" ${PROJECT_IS_TOP_LEVEL})
option(NEXORA_BUILD_EXAMPLES "Build examples" ${NEXORA_BUILD_TESTS})
option(NEXORA_BUILD_BENCHES "Build benchmarks" OFF)
option(NEXORA_BUILD_TOOLS "Build native probes" ${NEXORA_BUILD_TESTS})
option(NEXORA_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
option(NEXORA_INSTALL_PACKAGE "Install nexora package" ON)
find_program(NEXORA_CCACHE ccache)
if(NEXORA_CCACHE)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${NEXORA_CCACHE}")
endif()
include(NexoraCompilerOptions)
include(NexoraReflection)
include(NexoraSanitizers)
include(NexoraCompileFail)
include(NexoraCodegenGates)
nexora_detect_reflection()
include(NexoraTargets)
if(NEXORA_BUILD_TOOLS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/probes/CMakeLists.txt")
  add_subdirectory(src/probes)
endif()
if(NEXORA_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(NEXORA_BUILD_EXAMPLES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/CMakeLists.txt")
  add_subdirectory(examples)
endif()
if(NEXORA_BUILD_BENCHES)
  add_subdirectory(benches)
endif()
if(NEXORA_INSTALL_PACKAGE)
  install(TARGETS ${NEXORA_INTERFACE_TARGETS} nexora EXPORT nexora-targets)
  install(DIRECTORY include/nexora DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}" PATTERN "probes" EXCLUDE)
  install(EXPORT nexora-targets NAMESPACE nexora:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nexora")
  configure_package_config_file(cmake/nexora-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/nexora-config.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nexora")
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/nexora-config-version.cmake"
    VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nexora-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/nexora-config-version.cmake"
    cmake/NexoraTlsDependency.cmake DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nexora")
endif()
