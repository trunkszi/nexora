cmake_minimum_required(VERSION 4.0)
foreach(example IN ITEMS http1_keepalive_and_cares_dns http2_tls_multiplex_and_cancel websocket_echo_and_push_timer unified_gateway_server)
  add_executable(${example} ${example}.cc)
  nexora_configure_internal(${example})
  if(NEXORA_BUILD_TESTS)
    add_test(NAME ${example}_loopback COMMAND ${example})
    set_tests_properties(${example}_loopback PROPERTIES TIMEOUT 20 LABELS "e2e;loopback")
  endif()
endforeach()
