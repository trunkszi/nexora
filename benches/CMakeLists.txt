cmake_minimum_required(VERSION 4.0)
if(NOT NEXORA_BUILD_BENCHES)
  return()
endif()

add_executable(nexora_proto_hotpath_bench proto_hotpath_bench.cc)
target_link_libraries(nexora_proto_hotpath_bench PRIVATE nexora::http2)
uarch_configure_runtime(nexora_proto_hotpath_bench)
nexora_enable_warnings(nexora_proto_hotpath_bench)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime_hsm_bench.cc")
  if(NOT TARGET nexora_probes)
    message(FATAL_ERROR "runtime_hsm_bench requires NEXORA_BUILD_TOOLS=ON")
  endif()
  add_executable(runtime_hsm_bench "${CMAKE_CURRENT_SOURCE_DIR}/runtime_hsm_bench.cc")
  nexora_configure_internal(runtime_hsm_bench NO_SANITIZERS)
  target_link_libraries(runtime_hsm_bench PRIVATE nexora_probes)
  target_compile_options(runtime_hsm_bench PRIVATE -O3 -fno-lto)
  if(NEXORA_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME runtime_probe_oracles COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/scenario/probe_oracle_test.py" "$<TARGET_FILE:runtime_hsm_bench>")
  endif()

  add_executable(nexora_gem5_guest "${CMAKE_CURRENT_SOURCE_DIR}/runtime_hsm_bench.cc")
  nexora_configure_internal(nexora_gem5_guest NO_SANITIZERS)
  target_link_libraries(nexora_gem5_guest PRIVATE nexora_probes)
  target_include_directories(nexora_gem5_guest PRIVATE "${PROJECT_SOURCE_DIR}/src")
  target_compile_definitions(nexora_gem5_guest PRIVATE NEXORA_FREESTANDING=1)
  target_compile_options(nexora_gem5_guest PRIVATE -O3 -fno-lto -fno-stack-protector -fno-builtin -fno-pie)
  target_link_options(nexora_gem5_guest PRIVATE -fno-lto -nostdlib -static -Wl,-no-pie,--no-dynamic-linker)
  set_target_properties(nexora_gem5_guest PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(NEXORA_BUILD_TESTS)
    add_test(NAME gem5_guest_oracles COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/scenario/probe_oracle_test.py" "$<TARGET_FILE:nexora_gem5_guest>")
  endif()

  add_executable(nexora_gem5_roi_guest "${CMAKE_CURRENT_SOURCE_DIR}/runtime_hsm_bench.cc")
  nexora_configure_internal(nexora_gem5_roi_guest NO_SANITIZERS)
  target_link_libraries(nexora_gem5_roi_guest PRIVATE nexora_probes)
  target_include_directories(nexora_gem5_roi_guest PRIVATE "${PROJECT_SOURCE_DIR}/src")
  target_compile_definitions(nexora_gem5_roi_guest PRIVATE NEXORA_FREESTANDING=1 NEXORA_GEM5_ROI=1)
  target_compile_options(nexora_gem5_roi_guest PRIVATE -O3 -fno-lto -fno-stack-protector -fno-builtin -fno-pie)
  target_link_options(nexora_gem5_roi_guest PRIVATE -fno-lto -nostdlib -static -Wl,-no-pie,--no-dynamic-linker)
  set_target_properties(nexora_gem5_roi_guest PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endif()
